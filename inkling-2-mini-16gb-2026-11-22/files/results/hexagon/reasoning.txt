  for (const b of balls){
    for (let i=0;i<6;i++){
      const pa=v[i], pb=v[(i+1)%6];
      const abx=pb.x-pa.x, aby=pb.y-pa.y;
      const len2 = abx*abx+aby*aby;
      let t = ((b.x-pa.x)*abx + (b.y-pa.y)*aby)/len2;
      t = Math.max(0, Math.min(1, t));
      const qx = pa.x+abx*t, qy = pa.y+aby*t;
      let dx = b.x-qx, dy = b.y-qy;
