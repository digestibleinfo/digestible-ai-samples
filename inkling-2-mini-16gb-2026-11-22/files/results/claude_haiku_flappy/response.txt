<title>Flappy Bird</title>
<style>
  html, body {
    margin: 0;
    height: 100%;
    background: #1b1b1b;
    display: flex;
    align-items: center;
    justify-content: center;
    overflow: hidden;
    font-family: Arial, Helvetica, sans-serif;
  }
  canvas {
    background: #70c5ce;
    border: 2px solid #000;
    max-width: 100vw;
    max-height: 100vh;
    touch-action: none;
  }

<body>
<canvas id="game" width="400" height="600"></canvas>
<script>
(function () {
  const canvas = document.getElementById('game');
  const ctx = canvas.getContext('2d');
  const W = canvas.width;
  const H = canvas.height;

  const GROUND_H = 80;
  const GRAVITY = 0.45;
  const JUMP_V = -8;
  const PIPE_SPEED = 2.5;
  const PIPE_W = 60;
  const PIPE_GAP = 150;
  const PIPE_SPACING = 200;
  const BIRD_X = 90;
  const BIRD_R = 14;

  let bird, pipes, score, state, frame;

  function reset() {
    bird = { y: H / 2, vy: 0, rot: 0 };
    pipes = [];
    score = 0;
    frame = 0;
    state = 'ready';
  }

  function jump() {
    if (state === 'ready') {
      state = 'playing';
    }
    if (state === 'playing') {
      bird.vy = JUMP_V;
    } else if (state === 'over') {
      reset();
    }
  }

  function addPipe() {
    const minTop = 60;
    const maxTop = H - GROUND_H - PIPE_GAP - 60;
    const top = minTop + Math.random() * (maxTop - minTop);
    pipes.push({ x: W, top: top, passed: false });
  }

  function collidesWithPipe(p) {
    const birdLeft = BIRD_X - BIRD_R + 3;
    const birdRight = BIRD_X + BIRD_R - 3;
    const birdTop = bird.y - BIRD_R + 3;
    const birdBottom = bird.y + BIRD_R - 3;
    if (birdRight < p.x || birdLeft > p.x + PIPE_W) return false;
    return birdTop < p.top || birdBottom > p.top + PIPE_GAP;
  }

  function update() {
    if (state === 'playing') {
      frame++;
      bird.vy += GRAVITY;
      bird.y += bird.vy;
      bird.rot = Math.min(Math.PI / 2, Math.max(-0.45, bird.vy * 0.07));

      if (frame % 90 === 0) addPipe();

      for (const p of pipes) {
        p.x -= PIPE_SPEED;
        if (!p.passed && p.x + PIPE_W < BIRD_X - BIRD_R) {
          p.passed = true;
          score++;
        }
        if (collidesWithPipe(p)) {
          endGame();
        }
      }
      pipes = pipes.filter(p => p.x + PIPE_W > -10);

      if (bird.y + BIRD_R >= H - GROUND_H || bird.y - BIRD_R <= 0) {
        if (bird.y - BIRD_R <= 0) {
          bird.y = BIRD_R;
          bird.vy = 0;
        } else {
          endGame();
        }
      }
    } else if (state === 'ready') {
      bird.y = H / 2 + Math.sin(Date.now() / 200) * 8;
    }
  }

  function endGame() {
    if (state !== 'over') {
      state = 'over';
      bird.vy = 0;
    }
  }

  function drawBird() {
    ctx.save();
    ctx.translate(BIRD_X, bird.y);
    ctx.rotate(bird.rot);
    ctx.fillStyle = '#f7d51d';
    ctx.beginPath();
    ctx.arc(0, 0, BIRD_R, 0, Math.PI * 2);
    ctx.fill();
    ctx.strokeStyle = '#000';
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(5, -4, 5, 0, Math.PI * 2);
    ctx.fill();
    ctx.stroke();
    ctx.fillStyle = '#000';
    ctx.beginPath();
    ctx.arc(7, -4, 2, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#e8872a';
    ctx.beginPath();
    ctx.moveTo(BIRD_R - 2, -2);
    ctx.lineTo(BIRD_R + 8, 2);
    ctx.lineTo(BIRD_R - 2, 6);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();
    ctx.restore();
  }

  function drawPipe(p) {
    const bottomY = p.top + PIPE_GAP;
    ctx.fillStyle = '#73bf2e';
    ctx.strokeStyle = '#2d5a0e';
    ctx.lineWidth = 3;
    ctx.fillRect(p.x, 0, PIPE_W, p.top);
    ctx.strokeRect(p.x, -3, PIPE_W, p.top + 3);
    ctx.fillRect(p.x, bottomY, PIPE_W, H - GROUND_H - bottomY);
    ctx.strokeRect(p.x, bottomY, PIPE_W, H - GROUND_H - bottomY + 3);
    ctx.fillStyle = '#8fd848';
    ctx.fillRect(p.x - 5, p.top - 22, PIPE_W + 10, 22);
    ctx.strokeRect(p.x - 5, p.top - 22, PIPE_W + 10, 22);
    ctx.fillRect(p.x - 5, bottomY, PIPE_W + 10, 22);
    ctx.strokeRect(p.x - 5, bottomY, PIPE_W + 10, 22);
  }

  function drawGround() {
    ctx.fillStyle = '#ded895';
    ctx.fillRect(0, H - GROUND_H, W, GROUND_H);
    ctx.fillStyle = '#73bf2e';
    ctx.fillRect(0, H - GROUND_H, W, 20);
    ctx.strokeStyle = '#2d5a0e';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(0, H - GROUND_H);
    ctx.lineTo(W, H - GROUND_H);
    ctx.stroke();
  }

  function drawScore() {
    ctx.font = 'bold 48px Arial';
    ctx.textAlign = 'center';
    ctx.lineWidth = 6;
    ctx.strokeStyle = '#000';
    ctx.fillStyle = '#fff';
    ctx.strokeText(score, W / 2, 70);
    ctx.fillText(score, W / 2, 70);
  }

  function drawOverlayText(title, sub) {
    ctx.textAlign = 'center';
    ctx.font = 'bold 40px Arial';
    ctx.lineWidth = 6;
    ctx.strokeStyle = '#000';
    ctx.fillStyle = '#fff';
    ctx.strokeText(title, W / 2, H / 2 - 20);
    ctx.fillText(title, W / 2, H / 2 - 20);
    if (sub) {
      ctx.font = 'bold 22px Arial';
      ctx.lineWidth = 4;
      ctx.strokeText(sub, W / 2, H / 2 + 25);
      ctx.fillText(sub, W / 2, H / 2 + 25);
    }
  }

  function draw() {
    ctx.fillStyle = '#70c5ce';
    ctx.fillRect(0, 0, W, H);

    for (const p of pipes) drawPipe(p);
    drawGround();
    drawBird();

    if (state === 'playing' || state === 'over') {
      drawScore();
    }

    if (state === 'ready') {
      drawOverlayText('Flappy Bird', 'Press Space or Click to start');
    } else if (state === 'over') {
      drawOverlayText('Game Over', 'Score: ' + score + '  -  Press Space or Click to restart');
    }
  }

  function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
  }

  document.addEventListener('keydown', e => {
    if (e.code === 'Space') {
      e.preventDefault();
      if (!e.repeat) jump();
    }
  });

  canvas.addEventListener('mousedown', e => {
    e.preventDefault();
    jump();
  });

  canvas.addEventListener('touchstart', e => {
    e.preventDefault();
    jump();
  }, { passive: false });

  reset();
  loop();
})();
</script>
</body>
</html>
