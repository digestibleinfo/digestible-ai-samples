```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Flappy Bird</title>
<style>
  html, body {
    margin: 0;
    height: 100%;
    background: #1b1b1b;
    display: flex;
    align-items: center;
    justify-content: center;
    overflow: hidden;
    font-family: Arial, Helvetica, sans-serif;
  }
  canvas {
    background: #70c5ce;
    border: 2px solid #333;
    max-width: 100vw;
    max-height: 100vh;
    touch-action: manipulation;
  }
</style>
</head>
<body>
<canvas id="game" width="400" height="600"></canvas>
<script>
(function () {
  const canvas = document.getElementById('game');
  const ctx = canvas.getContext('2d');
  const W = canvas.width;
  const H = canvas.height;

  const GROUND_H = 80;
  const GRAVITY = 0.45;
  const JUMP_VEL = -8;
  const PIPE_W = 60;
  const PIPE_GAP = 150;
  const PIPE_SPEED = 2.5;
  const PIPE_INTERVAL = 1500; // ms
  const BIRD_X = 90;
  const BIRD_R = 14;

  let bird, pipes, score, state, lastPipeTime, lastTime, groundOffset;

  function reset() {
    bird = { y: H / 2, vy: 0, rot: 0 };
    pipes = [];
    score = 0;
    state = 'ready';
    lastPipeTime = 0;
    lastTime = performance.now();
    groundOffset = 0;
  }

  function jump() {
    if (state === 'ready') {
      state = 'playing';
      bird.vy = JUMP_VEL;
    } else if (state === 'playing') {
      bird.vy = JUMP_VEL;
    } else if (state === 'over') {
      reset();
    }
  }

  function spawnPipe() {
    const minTop = 60;
    const maxTop = H - GROUND_H - PIPE_GAP - 60;
    const top = minTop + Math.random() * (maxTop - minTop);
    pipes.push({ x: W, top: top, passed: false });
  }

  function collides(p) {
    const birdLeft = BIRD_X - BIRD_R + 4;
    const birdRight = BIRD_X + BIRD_R - 4;
    const birdTop = bird.y - BIRD_R + 4;
    const birdBottom = bird.y + BIRD_R - 4;
    if (birdRight > p.x && birdLeft < p.x + PIPE_W) {
      if (birdTop < p.top || birdBottom > p.top + PIPE_GAP) return true;
    }
    return false;
  }

  function update(dt) {
    if (state === 'ready') {
      bird.y = H / 2 + Math.sin(performance.now() / 200) * 8;
      groundOffset = (groundOffset + PIPE_SPEED) % 24;
      return;
    }
    if (state !== 'playing') return;

    bird.vy += GRAVITY;
    bird.y += bird.vy;
    bird.rot = Math.min(Math.PI / 2, Math.max(-0.4, bird.vy * 0.08));

    groundOffset = (groundOffset + PIPE_SPEED) % 24;

    const now = performance.now();
    if (now - lastPipeTime > PIPE_INTERVAL) {
      spawnPipe();
      lastPipeTime = now;
    }

    for (const p of pipes) {
      p.x -= PIPE_SPEED;
      if (!p.passed && p.x + PIPE_W < BIRD_X - BIRD_R) {
        p.passed = true;
        score++;
      }
      if (collides(p)) {
        endGame();
        return;
      }
    }
    pipes = pipes.filter(p => p.x + PIPE_W > -10);

    if (bird.y + BIRD_R >= H - GROUND_H || bird.y - BIRD_R <= 0) {
      endGame();
    }
  }

  function endGame() {
    state = 'over';
    bird.vy = 0;
  }

  function drawBird() {
    ctx.save();
    ctx.translate(BIRD_X, bird.y);
    ctx.rotate(bird.rot);
    ctx.fillStyle = '#f7d51d';
    ctx.beginPath();
    ctx.arc(0, 0, BIRD_R, 0, Math.PI * 2);
    ctx.fill();
    ctx.strokeStyle = '#333';
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(5, -4, 5, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#000';
    ctx.beginPath();
    ctx.arc(6, -4, 2, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = '#e8762b';
    ctx.beginPath();
    ctx.moveTo(BIRD_R - 2, -2);
    ctx.lineTo(BIRD_R + 8, 2);
    ctx.lineTo(BIRD_R - 2, 6);
    ctx.closePath();
    ctx.fill();
    ctx.restore();
  }

  function drawPipes() {
    for (const p of pipes) {
      ctx.fillStyle = '#4ec04e';
      ctx.fillRect(p.x, 0, PIPE_W, p.top);
      ctx.fillRect(p.x, p.top + PIPE_GAP, PIPE_W, H - GROUND_H - (p.top + PIPE_GAP));
      ctx.fillStyle = '#2e8b2e';
      ctx.fillRect(p.x - 4, p.top - 20, PIPE_W + 8, 20);
      ctx.fillRect(p.x - 4, p.top + PIPE_GAP, PIPE_W + 8, 20);
      ctx.strokeStyle = '#1e5e1e';
      ctx.lineWidth = 2;
      ctx.strokeRect(p.x, 0, PIPE_W, p.top);
      ctx.strokeRect(p.x, p.top + PIPE_GAP, PIPE_W, H - GROUND_H - (p.top + PIPE_GAP));
    }
  }

  function drawGround() {
    ctx.fillStyle = '#ded895';
    ctx.fillRect(0, H - GROUND_H, W, GROUND_H);
    ctx.fillStyle = '#73bf2e';
    ctx.fillRect(0, H - GROUND_H, W, 16);
    ctx.fillStyle = '#5a9a20';
    for (let x = -24 + (groundOffset % 24); x < W; x += 24) {
      ctx.fillRect(x, H - GROUND_H + 4, 12, 4);
    }
  }

  function drawScore() {
    ctx.font = 'bold 44px Arial';
    ctx.textAlign = 'center';
    ctx.lineWidth = 6;
    ctx.strokeStyle = '#000';
    ctx.fillStyle = '#fff';
    ctx.strokeText(score, W / 2, 70);
    ctx.fillText(score, W / 2, 70);
  }

  function drawOverlay(title, sub) {
    ctx.fillStyle = 'rgba(0,0,0,0.4)';
    ctx.fillRect(0, 0, W, H);
    ctx.textAlign = 'center';
    ctx.font = 'bold 42px Arial';
    ctx.lineWidth = 6;
    ctx.strokeStyle = '#000';
    ctx.fillStyle = '#fff';
    ctx.strokeText(title, W / 2, H / 2 - 40);
    ctx.fillText(title, W / 2, H / 2 - 40);
    ctx.font = 'bold 24px Arial';
    ctx.lineWidth = 4;
    ctx.strokeText(sub, W / 2, H / 2 + 10);
    ctx.fillText(sub, W / 2, H / 2 + 10);
  }

  function draw() {
    ctx.clearRect(0, 0, W, H);
    ctx.fillStyle = '#70c5ce';
    ctx.fillRect(0, 0, W, H);
    drawPipes();
    drawGround();
    drawBird();
    if (state === 'playing') {
      drawScore();
    } else if (state === 'ready') {
      drawScore();
      ctx.textAlign = 'center';
      ctx.font = 'bold 22px Arial';
      ctx.lineWidth = 4;
      ctx.strokeStyle = '#000';
      ctx.fillStyle = '#fff';
      ctx.strokeText('Press Space or Click to Start', W / 2, H / 2 + 60);
      ctx.fillText('Press Space or Click to Start', W / 2, H / 2 + 60);
    } else if (state === 'over') {
      drawScore();
      drawOverlay('Game Over', 'Score: ' + score);
      ctx.font = 'bold 18px Arial';
      ctx.lineWidth = 3;
      ctx.strokeText('Press Space or Click to Restart', W / 2, H / 2 + 50);
      ctx.fillText('Press Space or Click to Restart', W / 2, H / 2 + 50);
    }
  }

  function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
  }

  document.addEventListener('keydown', e => {
    if (e.code === 'Space') {
      e.preventDefault();
      jump();
    }
  });
  canvas.addEventListener('mousedown', jump);
  canvas.addEventListener('touchstart', e => { e.preventDefault(); jump(); }, { passive: false });

  reset();
  loop();
})();
</script>
</body>
</html>
```
